Connected sheet

This page reads from the To Do tab in your existing Google Sheet. To activate the live connection, open the sheet, go to File › Share › Publish to the web, select To Do (the specific tab) and Comma-separated values (.csv), click Publish, then click Connect below.

The To Do tab needs these columns in row 1 (A–F):
Date Added  ·  Responsible  ·  Topic  ·  Task Name  ·  Description  ·  Date Completed

To write completions back to the sheet automatically, open Extensions › Apps Script in your sheet, paste the code below, and deploy it as a web app (Execute as: Me · Who has access: Anyone). Paste that deployment URL into the field above instead of the CSV URL. 

function doGet(e) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('To Do') || ss.getActiveSheet();
  const data  = sheet.getDataRange().getValues();
  const keys  = data[0];
  const rows  = data.slice(1).map((r, i) =>
    Object.fromEntries(keys.map((k, j) => [k, { value: r[j], row: i + 2 }]))
  );
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const { row, date } = JSON.parse(e.postData.contents);
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('To Do') || ss.getActiveSheet();
  sheet.getRange(row, 6).setValue(date); // column F = Date Completed
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
