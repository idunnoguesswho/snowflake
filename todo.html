Connected sheet — Tasks tab

This page reads from the Tasks tab in your family Google Sheet. To activate the live connection, open the sheet, go to File › Share › Publish to the web, select Tasks (the specific tab) and Comma-separated values (.csv), click Publish, copy the URL, paste it below, then click Connect.

The Tasks tab columns (already set up in your sheet):
ID  ·  Date Added  ·  Responsible  ·  Category  ·  Est Time  ·  Task Name EN  ·  Task Name FR  ·  Description EN  ·  Description FR  ·  Date Completed

To write completions back to the sheet automatically, open Extensions › Apps Script in your sheet, paste the code below, and deploy it as a web app (Execute as: Me · Who has access: Anyone). Paste that deployment URL into the field above instead of the CSV URL. 

function doGet(e) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Tasks');
  const data  = sheet.getDataRange().getValues();
  const keys  = data[0];
  const rows  = data.slice(1).map((r, i) =>
    Object.fromEntries(keys.map((k, j) => [k, { value: r[j], row: i + 2 }]))
  );
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const { row, date } = JSON.parse(e.postData.contents);
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Tasks');
  sheet.getRange(row, 10).setValue(date); // column J = Date Completed
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
